#!/usr/bin/env python3
"""
Native audio capture for Mac host.
Saves audio files to shared volume for Docker processing.
"""
import pyaudio
import webrtcvad
import wave
import time
import os
from datetime import datetime
import numpy as np

class HostAudioCapture:
    def __init__(self):
        self.sample_rate = 16000
        self.chunk_size = 1024
        self.vad = webrtcvad.Vad(2)
        
        # Ensure output directory exists
        os.makedirs("data/audio", exist_ok=True)
        
    def start_capture(self):
        """Start continuous audio capture"""
        p = pyaudio.PyAudio()
        
        stream = p.open(
            format=pyaudio.paInt16,
            channels=1,
            rate=self.sample_rate,
            input=True,
            frames_per_buffer=self.chunk_size
        )
        
        print("🎤 Starting audio capture...")
        print("   Files will be saved to ./data/audio/")
        print("   Docker containers will process them automatically")
        print("   Press Ctrl+C to stop")
        
        try:
            segment_frames = []
            segment_start = time.time()
            
            while True:
                data = stream.read(self.chunk_size)
                segment_frames.append(data)
                
                # Save segment every 30 seconds or on silence
                if time.time() - segment_start >= 30:
                    self.save_segment(segment_frames)
                    segment_frames = []
                    segment_start = time.time()
                    
        except KeyboardInterrupt:
            print("\n🛑 Stopping audio capture...")
        finally:
            stream.stop_stream()
            stream.close()
            p.terminate()
            
            # Save any remaining audio
            if segment_frames:
                self.save_segment(segment_frames)
    
    def save_segment(self, frames):
        """Save audio segment to shared volume"""
        if not frames:
            return
            
        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S_%f")[:-3]
        filename = f"data/audio/segment_{timestamp}.wav"
        
        # Write WAV file
        with wave.open(filename, 'wb') as wf:
            wf.setnchannels(1)
            wf.setsampwidth(2)  # 2 bytes = 16 bit
            wf.setframerate(self.sample_rate)
            wf.writeframes(b''.join(frames))
        
        print(f"📁 Saved: {filename}")

if __name__ == "__main__":
    capture = HostAudioCapture()
    capture.start_capture()
EOF

# Make it executable

Intelligent Audio Segmentation

💡
Smart Segmentation: Instead of fixed 30-second chunks, Mike Logger uses intelligent segmentation based on speech patterns, silence detection, and natural pauses.

Segmentation Strategies

┌─────────────────────────────────────────────────────────────────────────┐
│                     INTELLIGENT AUDIO SEGMENTATION                      │
├─────────────────────────────────────────────────────────────────────────┤
│                                                                         │
│  1. Voice Activity Detection (VAD)                                      │
│     ├─ Detect speech vs silence in real-time                          │
│     ├─ Buffer speech with padding (300ms before/after)                │
│     └─ Trigger segmentation on extended silence (2+ seconds)          │
│                                                                         │
│  2. Natural Pause Detection                                             │
│     ├─ Short pause (0.5-1s): Possible sentence boundary               │
│     ├─ Medium pause (1-2s): Likely paragraph boundary                 │
│     └─ Long pause (2s+): Definite segment boundary                    │
│                                                                         │
│  3. Acoustic Feature Analysis                                           │
│     ├─ Pitch contour: Falling pitch = sentence end                    │
│     ├─ Energy envelope: Decreasing energy = phrase end                │
│     └─ Spectral features: Identify breath sounds                      │
│                                                                         │
│  4. Time-Based Constraints                                              │
│     ├─ Minimum segment: 5 seconds (avoid tiny fragments)              │
│     ├─ Maximum segment: 2 minutes (prevent huge files)                │
│     └─ Overlap window: 500ms (ensure no lost words)                   │
│                                                                         │
└─────────────────────────────────────────────────────────────────────────┘

Implement Smart Audio Segmentation

# src/smart_audio_capture.py
import collections
import webrtcvad
import pyaudio
import numpy as np
from scipy import signal
import time

class SmartAudioCapture:
    def __init__(self):
        self.vad = webrtcvad.Vad(2)  # Aggressiveness 0-3
        self.sample_rate = 16000
        self.frame_duration_ms = 30  # VAD frame size
        self.frames_per_buffer = int(self.sample_rate * self.frame_duration_ms / 1000)
        
        # Segmentation parameters
        self.min_segment_duration = 5.0  # seconds
        self.max_segment_duration = 120.0  # seconds
        self.silence_threshold = 1.5  # seconds to trigger split
        self.speech_pad_ms = 300  # padding around speech
        
        # Ring buffer for smooth segmentation
        self.ring_buffer = collections.deque(maxlen=int(self.silence_threshold * 1000 / self.frame_duration_ms))
        
    def is_speech(self, frame):
        """Check if frame contains speech using VAD"""
        return self.vad.is_speech(frame, self.sample_rate)
    
    def detect_natural_pause(self, audio_buffer):
        """Analyze audio for natural speech boundaries"""
        # Convert to numpy array
        audio_data = np.frombuffer(b''.join(audio_buffer), dtype=np.int16)
        
        # Calculate energy envelope
        energy = np.abs(audio_data)
        smoothed_energy = signal.savgol_filter(energy, 1001, 3)
        
        # Find valleys (potential pause points)
        valleys = signal.find_peaks(-smoothed_energy, distance=self.sample_rate//2)[0]
        
        # Score each valley based on:
        # 1. Depth of energy drop
        # 2. Duration of low energy
        # 3. Distance from start/end
        pause_scores = []
        for valley in valleys:
            if valley < len(audio_data) * 0.1 or valley > len(audio_data) * 0.9:
                continue  # Too close to edges
                
            # Calculate pause metrics
            depth = 1.0 - (smoothed_energy[valley] / np.mean(smoothed_energy))
            duration = self._measure_pause_duration(smoothed_energy, valley)
            
            score = depth * duration
            pause_scores.append((valley, score))
        
        # Return best pause point
        if pause_scores:
            best_pause = max(pause_scores, key=lambda x: x[1])
            return best_pause[0] / self.sample_rate
        
        return None
    
    def _measure_pause_duration(self, energy, center):
        """Measure duration of low energy around a point"""
        threshold = np.mean(energy) * 0.3
        
        # Find pause boundaries
        left = center
        while left > 0 and energy[left] < threshold:
            left -= 1
            
        right = center
        while right < len(energy) and energy[right] < threshold:
            right += 1
            
        return (right - left) / self.sample_rate
    
    def segment_audio_stream(self, audio_callback):
        """Main segmentation loop with intelligent boundaries"""
        p = pyaudio.PyAudio()
        stream = p.open(
            format=pyaudio.paInt16,
            channels=1,
            rate=self.sample_rate,
            input=True,
            frames_per_buffer=self.frames_per_buffer
        )
        
        current_segment = []
        segment_start_time = time.time()
        triggered = False
        
        try:
            while True:
                frame = stream.read(self.frames_per_buffer)
                is_speech = self.is_speech(frame)
                
                # Update ring buffer
                self.ring_buffer.append(is_speech)
                
                # Add frame to current segment
                current_segment.append(frame)
                
                # Check for segment boundaries
                segment_duration = time.time() - segment_start_time
                
                # Force segmentation at max duration
                if segment_duration >= self.max_segment_duration:
                    self._save_segment(current_segment, audio_callback)
                    current_segment = []
                    segment_start_time = time.time()
                    triggered = False
                    
                # Check for natural segmentation point
                elif segment_duration >= self.min_segment_duration:
                    # Count recent speech frames
                    num_speech = sum(self.ring_buffer)
                    
                    # If mostly silence, look for natural pause
                    if num_speech < len(self.ring_buffer) * 0.1:
                        if not triggered:
                            # Find best cut point
                            pause_point = self.detect_natural_pause(current_segment[-int(self.sample_rate*2):])
                            
                            if pause_point:
                                # Cut at natural pause
                                cut_index = len(current_segment) - int((2.0 - pause_point) * self.sample_rate / self.frames_per_buffer)
                                
                                # Save segment up to pause
                                self._save_segment(current_segment[:cut_index], audio_callback)
                                
                                # Keep remainder for next segment
                                current_segment = current_segment[cut_index:]
                                segment_start_time = time.time()
                                triggered = False
                            else:
                                triggered = True
                    else:
                        triggered = False
                        
        finally:
            stream.stop_stream()
            stream.close()
            p.terminate()
    
    def _save_segment(self, frames, callback):
        """Save audio segment with metadata"""
        if len(frames) < self.sample_rate * self.min_segment_duration / self.frames_per_buffer:
            return  # Too short
            
        audio_data = b''.join(frames)
        
        # Calculate segment statistics
        np_data = np.frombuffer(audio_data, dtype=np.int16)
        
        metadata = {
            'duration': len(np_data) / self.sample_rate,
            'energy_mean': float(np.mean(np.abs(np_data))),
            'energy_std': float(np.std(np.abs(np_data))),
            'zero_crossings': np.sum(np.diff(np.sign(np_data)) != 0),
            'timestamp': time.time()
        }
        
        callback(audio_data, metadata)

Troubleshooting

Common Issues and Solutions

🔧
No Audio Input Detected
• Check microphone permissions in system settings
• Run python test_audio.py to list devices
• Specify device index: --audio-device 2
🔧
Whisper Running Slowly
• Use smaller model: --whisper-model tiny
• Enable GPU: pip install torch torchvision torchaudio --index-url https://download.pytorch.org/whl/cu118
• Reduce sample rate: --sample-rate 8000
🔧
High Memory Usage
• Reduce buffer sizes in config
• Use smaller Whisper model
• Enable swap space on Linux
• Set memory limits in Docker

Whisper (base) ~40% 1GB Use GPU or smaller model
